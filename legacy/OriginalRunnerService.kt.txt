package com.example.htmlbackgroundrunner

import android.app.*
import android.content.Context
import android.content.Intent
import android.os.Build
import android.os.IBinder

class RunnerService : Service() {
    companion object {
        const val ACTION_START = "runner.START"
        const val ACTION_STOP = "runner.STOP"
        const val CHANNEL_ID = "html_runner_status"
        const val NOTIFICATION_ID = 41
        @Volatile private var active = false
        fun start(context: Context) {
            val i = Intent(context, RunnerService::class.java).setAction(ACTION_START)
            if (Build.VERSION.SDK_INT >= 26) context.startForegroundService(i) else context.startService(i)
        }
        fun stop(context: Context) {
            context.startService(Intent(context, RunnerService::class.java).setAction(ACTION_STOP))
        }
        fun isRunning(context: Context) = active
    }

    override fun onCreate() {
        super.onCreate()
        createChannel()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                active = false
                getSharedPreferences("runner", MODE_PRIVATE).edit().putBoolean("running", false).apply()
                stopForeground(STOP_FOREGROUND_REMOVE)
                stopSelf()
                return START_NOT_STICKY
            }
            else -> {
                active = true
                getSharedPreferences("runner", MODE_PRIVATE).edit().putBoolean("running", true).apply()
                startForeground(NOTIFICATION_ID, notification())
                return START_STICKY
            }
        }
    }

    private fun createChannel() {
        if (Build.VERSION.SDK_INT >= 26) {
            val channel = NotificationChannel(CHANNEL_ID, "HTML runner", NotificationManager.IMPORTANCE_LOW)
            channel.description = "Shows when an HTML runner session is active"
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }
    }

    private fun notification(): Notification {
        val open = PendingIntent.getActivity(
            this, 1, Intent(this, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_SINGLE_TOP),
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val stop = PendingIntent.getService(
            this, 2, Intent(this, RunnerService::class.java).setAction(ACTION_STOP),
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        return Notification.Builder(this, CHANNEL_ID)
            .setSmallIcon(android.R.drawable.ic_media_play)
            .setContentTitle("HTML Runner — Running")
            .setContentText("Your HTML project session is active")
            .setContentIntent(open)
            .setOngoing(true)
            .addAction(Notification.Action.Builder(null, "Open", open).build())
            .addAction(Notification.Action.Builder(null, "Stop", stop).build())
            .build()
    }

    override fun onDestroy() {
        active = false
        getSharedPreferences("runner", MODE_PRIVATE).edit().putBoolean("running", false).apply()
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onTaskRemoved(rootIntent: Intent?) {
        // START_STICKY permits Android to recreate the service where system policy allows.
        super.onTaskRemoved(rootIntent)
    }
}
